cmake_minimum_required(VERSION 3.22.1)
project(media_mpv)
add_library(mpv SHARED IMPORTED)
set_target_properties(mpv PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libmpv.so)
add_library(avcodec SHARED IMPORTED)
set_target_properties(avcodec PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libavcodec.so)
add_library(media_mpv SHARED engine.cpp)
target_include_directories(media_mpv PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(media_mpv mpv avcodec android log)
target_link_options(media_mpv PRIVATE -Wl,-z,max-page-size=16384)
